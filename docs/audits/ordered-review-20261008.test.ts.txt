afterEach(() => { cleanup(); localStorage.clear(); setActiveMap('village'); useGame.setState({ modal: null, trip: null }) })

describe('전체 코드 리뷰: 추가 재현', () => {
  it('새 터 책상도 스페이스와 조이스틱 앞칸 상호작용 대상으로 잡혀야 한다', () => {
    const base = newGame(CONTENT)
    const s: GameState = { ...base, map: 'newland', scenes: [], flags: { ...base.flags, newlandGift: 1 },
      player: { ...base.player, ...INTERIOR_DESK.stand, facing: 'up', path: [] } }
    syncHome(s)
    expect(tapTile(s, INTERIOR_DESK.tile).target).toMatchObject({ kind: 'place', id: 'otDesk' })
    expect({ space: interactTile(s), joystick: pressTile(s) }).toEqual({ space: INTERIOR_DESK.tile, joystick: INTERIOR_DESK.tile })
  })

  it('저장을 불러와도 퍼즐로 마친 절 수와 시간·피로 비율이 유지되어야 한다', () => {
    const base = newGame(CONTENT)
    const key = careKey('mt', 1)
    const s = { ...base, copyCare: { at: key, typos: 0, puzzle: true, puzzled: 6 } }
    const back = deserialize(serialize(s), CONTENT)!
    expect(back).not.toBeNull()
    expect(chapterEffort(back.copyCare, key, 25)).toBeCloseTo(chapterEffort(s.copyCare, key, 25))
    expect(back.copyCare?.puzzled).toBe(6)
  })

  it('다른 책을 쓰고 돌아와도 이미 퍼즐로 쓴 장에 직접쓰기 정성 도장이 생기지 않아야 한다', () => {
    useGame.setState({ game: { ...newGame(CONTENT), scenes: [] }, modal: null })
    useGame.getState().copyBook('mt')
    expect(useGame.getState().copyPuzzle()).toBe(true)
    expect(useGame.getState().game.copyCare?.puzzle).toBe(true)
    useGame.getState().copyBook('mk')
    expect(useGame.getState().copyPuzzle()).toBe(true)
    useGame.getState().copyBook('mt')
    for (let i = 0; i < 30 && !useGame.getState().game.progress.mt.completed.includes(1); i++) {
      const at = copySpot(useGame.getState().game, 'mt', CONTENT)!
      expect(useGame.getState().copyType(at.verse.text)).toBe(true)
    }
    expect(useGame.getState().game.progress.mt.completed).toContain(1)
    expect(useGame.getState().game.careDone ?? []).not.toContain('mt:1')
  })

  it('자동 저장에서 도감·업적을 기록하면 실행 중 상태에도 반영되어야 한다', () => {
    setActiveSlot('1')
    const s = { ...newGame(CONTENT), scenes: [], coins: 500, achieved: [] }
    useGame.setState({ game: s, modal: { kind: 'cooking' }, clockMs: 0, award: null, trip: null })
    useGame.getState().frame(20)
    const saved = deserialize(localStorage.getItem(SAVE_KEY), CONTENT)!
    expect(saved.achieved.some(a => a.id === 'coins500')).toBe(true)
    expect(useGame.getState().game.achieved.some(a => a.id === 'coins500')).toBe(true)
    expect(recordProgress(useGame.getState().game).fresh).toEqual([])
  })

  it('현재 버전 저장의 사용자 메모는 옛 주민 id 마이그레이션으로 바뀌지 않아야 한다', () => {
    const base = newGame(CONTENT)
    const text = 'innkeeper, lambFound: 내가 직접 쓴 메모'
    const s = { ...base, myLines: { [CONTENT.pieces[0].id]: text } }
    const back = deserialize(serialize(s), CONTENT)!
    expect(back.myLines[CONTENT.pieces[0].id]).toBe(text)
  })

  it('식탁을 다 비운 뒤에도 반응에는 실제로 먹은 요리 이름이 남아야 한다', () => {
    const base = newGame(CONTENT)
    let s: GameState = { ...base, scenes: [], inv: { figPlate: 2 }, clock: { day: 3, minute: 900 },
      player: { ...base.player, x: 19, y: 113, path: [] } }
    syncHome(s)
    let room: Furniture[] = [...s.room]
    for (const [item, x, y, facing] of [['table', 19, 112, undefined], ['chair', 19, 113, 'up'], ['chair', 20, 113, 'up']] as const) {
      const f = placement(room, item, { x, y }, facing)!
      expect(f).toBeTruthy()
      room = [...room, f]
      syncHome({ ...s, room })
    }
    s = setSpace({ ...s, room }, room.find(f => f.item === 'table')!, 'tea')!
    s = { ...s, npcs: { ...s.npcs, poppy: { ...s.npcs.poppy, visible: true, x: 21, y: 112, path: [] } } }
    s = serveTable(s, 'figPlate', 2)
    expect(cookOf(s).table?.left).toBe(2)
    useGame.setState({ game: s, modal: { kind: 'cooking' } })
    const ui = render(createElement(CookingView))
    act(() => fireEvent.click(ui.getByText(T.cooking.toTable)))
    act(() => fireEvent.click(ui.getByRole('checkbox')))
    act(() => fireEvent.click(ui.getByText(fill(T.cooking.sitWith, { n: 1 }))))
    expect(cookOf(useGame.getState().game).table).toBeUndefined()
    expect(ui.container.textContent).toContain(itemName('figPlate'))
    expect(ui.container.textContent).not.toContain(itemName('honeyBread'))
  })
})

describe('목록 순서 추가 경계 재현', () => {
  it('17: 주민이 이사 와도 이미 게시된 부탁의 사람과 내용은 바뀌지 않는다', () => {
    const base = newGame(CONTENT)
    const before = { ...base, clock: { day: 10, minute: 660 }, flags: { ...base.flags, villageLevel: 0 } }
    const after = { ...before, flags: { ...before.flags, villageLevel: 3, 'movedIn:carpenter': 1 } }
    const old = copyRequestsToday(before, CONTENT)
    const fresh = copyRequestsToday(after, CONTENT)
    const changed = old.filter(r => JSON.stringify(r) !== JSON.stringify(fresh.find(n => n.id === r.id)))
    console.log('request pool change', changed.slice(0, 2).map(r => ({ before: r, after: fresh.find(n => n.id === r.id) })))
    expect(changed).toEqual([])
  })

  it('19: 이미 주민 배우자가 있는 사람에게 플레이어가 고백할 수 없다', () => {
    const base = newGame(CONTENT)
    const gen = newGenState(CONTENT, 7, 50)
    const hid = 'h1', rid = relationId('rudy', 'poppy')
    gen.persons.rudy = { ...gen.persons.rudy, spouse: 'poppy', household: hid }
    gen.persons.poppy = { ...gen.persons.poppy, spouse: 'rudy', household: hid }
    gen.households[hid] = { id: hid, members: ['rudy', 'poppy'], home: 'rudy', children: [], since: 20, lastBirth: null }
    gen.relations[rid] = { id: rid, a: 'poppy', b: 'rudy', stage: 'spouse', since: 20 }
    const s = { ...base, gen, scenes: [], clock: { day: 50, minute: 660 }, hearts: { ...base.hearts, rudy: 100 }, inv: { bouquet: 1 } }
    const def = CONTENT.neighbors.find(n => n.id === 'rudy')!
    const next = giveBouquet(s, def)
    console.log('married confession', { block: canGiveBouquet(s, def), playerPartner: next.romance.partner, npcSpouse: next.gen?.persons.rudy.spouse })
    expect(canGiveBouquet(s, def)).not.toBeNull()
    expect(next.romance.partner).toBeNull()
  })

  it('22: 주민 정원이 차면 출생 예정은 취소되지 않고 대기한다', () => {
    const gen = newGenState(CONTENT, 7, 50)
    const hid = 'h1'
    gen.households[hid] = { id: hid, members: ['rudy', 'poppy'], home: 'rudy', children: [], since: 10, lastBirth: null, cribAsk: { day: 15, given: true }, birthDue: 50 }
    for (let i = 1; i <= MAX_GENERATED; i++) gen.persons[`p${i}`] = { id: `p${i}`, origin: 'born', born: 1, stage: 'child', look: 'f', name: `아이${i}`, parents: [], spouse: null, household: null }
    const next = settleBirth(gen, 50, { partner: null, busy: () => false })
    console.log('birth at capacity', { before: gen.households[hid].birthDue, after: next.households[hid].birthDue })
    expect(next.households[hid].birthDue).toBe(50)
  })

  it('23: 따라갈 목표가 현재 동물 자리이면 지난 경로를 멈춘다', () => {
    const at = { x: 12, y: 18 }
    const c = { ...adopt('dog', '누렁이', 1, at), path: [{ x: 12, y: 19 }, { x: 12, y: 20 }] }
    const next = stepCompanion(c, at, 0.25)
    expect({ x: next.x, y: next.y, path: next.path }).toEqual({ ...at, path: [] })
  })

  it('23: 새 터에서 가족 메뉴로 따라오기를 눌러도 동물의 첫 마을 좌표를 유지한다', () => {
    const base = newGame(CONTENT)
    const at = { x: 19, y: 114 }
    const s: GameState = { ...base, map: 'newland', scenes: [], flags: { ...base.flags, newlandGift: 1 }, player: { ...base.player, ...INTERIOR_DESK.stand }, companion: { ...adopt('cat', '나비', 1, at), stay: true } }
    syncHome(s)
    const next = setCompanionStay(s, false)
    console.log('pet map coords', { before: at, after: { x: next.companion!.x, y: next.companion!.y } })
    expect({ x: next.companion!.x, y: next.companion!.y }).toEqual(at)
  })

  it('27: 빈집 주민 자리에 나무가 있으면 입주시키지 않거나 안전한 자리를 고른다', () => {
    const done = (id: string, kind: Build['kind'], x: number, slot?: number): Build => ({ id, kind, x, y: 12, facing: 'down', state: 'done', orderedDay: 1, paid: { coins: 0, items: {} }, refunded: false, ...(slot === undefined ? {} : { slot }) })
    const house = done('b2', 'home', 16, 0)
    const base = newGame(CONTENT)
    let s: GameState = { ...base, map: 'newland', scenes: [], coins: 2000, clock: { day: 50, minute: 660 }, player: { ...base.player, x: 4, y: 20, path: [] },
      flags: { ...base.flags, newlandGift: 1, newlandRevealed: 1, guestNow: 1, guestSince: 50, 'guestGift:nelly': 40 }, newland: { builds: [done('b1', 'guest', 10), house], tiles: {}, nextId: 3, settledDay: 50 } }
    syncHome(s)
    const at = settledSpot(house)!
    expect(canPlace(s, 'treeSmall', at.x, at.y)).toBeNull()
    s = orderBuild(s, 'treeSmall', at.x, at.y)!
    expect(s).not.toBeNull()
    syncHome(s)
    const invited = inviteGuest(s)
    console.log('blocked vacant-home invite', { at, block: canInviteGuest(s), living: invited.flags['settled:nelly'], walkable: isWalkable(at) })
    expect(canInviteGuest(s) === null || isWalkable(at)).toBe(true)
  })

  it('25: 배운 마감으로 놓은 가구는 저장 왕복 뒤에도 모습을 유지한다', () => {
    const base = newGame(CONTENT)
    const s = { ...base, scenes: [], inv: { stool: 1 }, flags: { ...base.flags, 'skillFinish:stool': 2 }, player: { ...base.player, x: 19, y: 113, path: [] } }
    const placed = placeFurniture(s, 'stool', { x: 18, y: 114 })!
    expect(placed).not.toBeNull()
    expect(placed.room.find(f => f.item === 'stool')?.finish).toBe('warm')
    const back = deserialize(serialize(placed), CONTENT)!
    expect(back.room.find(f => f.item === 'stool')?.finish).toBe('warm')
  })

  it('25: 가구를 회전해도 선택한 마감이 유지된다', () => {
    const base = newGame(CONTENT)
    const s = { ...base, scenes: [], inv: { stool: 1 }, flags: { ...base.flags, 'skillFinish:stool': 2 }, player: { ...base.player, x: 19, y: 113, path: [] } }
    const placed = placeFurniture(s, 'stool', { x: 18, y: 114 })!
    const piece = placed.room.find(f => f.item === 'stool')!
    expect(piece.finish).toBe('warm')
    const turned = rotateFurniture(placed, piece)!
    expect(turned).not.toBeNull()
    expect(turned.room.find(f => f.item === 'stool')?.finish).toBe('warm')
  })

  it('29: 현재 게시판 부탁 열 번을 끝내면 게시판 단골 업적이 열린다', () => {
    const base = newGame(CONTENT)
    const piece = CONTENT.pieces[0]
    let s = { ...base, scenes: [], collected: [piece.id] }
    for (let day = 1; day <= 5; day++) {
      s = { ...s, clock: { day, minute: 660 } }
      for (const r of requestsOn(day, ['baker', 'child'])) {
        const done = writeRequest(s, r, piece.id, { moods: [], chars: 20, care: false }, CONTENT)
        expect(done).not.toBeNull()
        s = done!.state
      }
    }
    expect(Object.keys(s.flags).filter(k => k.startsWith('req:'))).toHaveLength(10)
    expect(recordProgress(s).state.achieved.map(a => a.id)).toContain('board10')
  })
})
