function fest(b: GameState, task?: { stage: number; selection: string; choice: string }, present: string[] = []): GameState {
  const npcs = Object.fromEntries(Object.entries(b.npcs).map(([id, n]) => [id, { ...n, visible: false }]))
  for (const id of present) {
    // eslint-disable-next-line @typescript-eslint/no-explicit-any
    const spot = FESTIVAL_SPOTS[id as keyof typeof FESTIVAL_SPOTS]
    npcs[id] = { ...b.npcs[id], ...spot, visible: true, path: [] }
  }
  return { ...b, clock: { day: 20, minute: 1090 }, player: { ...b.player, ...FIRE }, npcs, plans: { appts: [], nextId: 1 }, clubSessions: {},
    fests: [{ id: 'fest:1', kind: 'festival', day: 20, place: 'bench', members: [], role: 'deco', reserve: {}, step: 'tidy', joined: false, consumed: false, closed: false, cancelled: false, ...(task ? { task } : {}) } as never] }
}

    expect(clubVenues(s, 'garden')).toContain('garden')
    expect(clubVenues(s, 'tea')).not.toContain('garden')

    fireEvent.click(screen.getByText(C.nextButton))
    fireEvent.click(screen.getByText(T.plans.place.garden))
    fireEvent.click(screen.getByText(C.back))
    fireEvent.click(screen.getByText(T.plans.activity.tea))
    fireEvent.click(screen.getByText(C.nextButton)) // step1: no place highlighted, but next is enabled
    const next = screen.getByText(C.nextButton) as HTMLButtonElement
    expect(next.disabled).toBe(false)
    expect(document.querySelectorAll('.actions.menu.column button.primary').length).toBe(0)
    fireEvent.click(next)
    // members
    const box = document.querySelector('input[type="checkbox"]:not([disabled])') as HTMLInputElement
    fireEvent.click(box)
    fireEvent.click(screen.getByText(C.nextButton))
    fireEvent.click(screen.getByText(C.nextButton))
    fireEvent.click(screen.getByText(C.save))
    expect(screen.getByRole('alert').textContent).toBe(C.errors.venue)
  })

  it('D Family: 진로 고르기 선택이 아이를 바꿔도 남아 다른 아이 진로로 확정된다', () => {
    const b = newGame(CONTENT)
    const kid = (name: string, born: number, job: string) => ({ name, look: 'boy', born, stats: b.stats, job, jobConfirmed: false, left: false })
    const g: GameState = { ...b, clock: { day: 300, minute: 600 }, child: kid('가', 10, 'scribe') as never, siblings: [kid('나', 20, 'fisher')] as never }
    useGame.setState({ game: g, modal: { kind: 'family' } as never })
    render(h(Family))
    const sel = document.querySelector('select') as HTMLSelectElement
    fireEvent.change(sel, { target: { value: 'merchant' } })
    fireEvent.click(screen.getByRole('button', { name: /나/ }))
    expect(useGame.getState().game.child?.name).toBe('나')
    expect((document.querySelector('select') as HTMLSelectElement).value).toBe('merchant')
