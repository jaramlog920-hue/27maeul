function mulberry32(a: number) {
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0
    let t = Math.imul(a ^ (a >>> 15), 1 | a)
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
  }
}

const dupes = (a: readonly unknown[] | undefined) => {
  if (!Array.isArray(a)) return []

  for (const x of a) { const k = typeof x === 'string' ? x : JSON.stringify(x); if (seen.has(k)) d.push(k); seen.add(k) }

function canon(x: unknown): unknown {
  if (Array.isArray(x)) return x.map(canon)
  if (x && typeof x === 'object') {
    const o: Record<string, unknown> = {}
    for (const k of Object.keys(x as object).sort()) {
      const v = (x as Record<string, unknown>)[k]
      if (v === undefined) continue
      o[k] = canon(v)
    }
    return o
  }
  return x
}

function diffKeys(a: Record<string, unknown>, b: Record<string, unknown>): string[] {
  const out: string[] = []
  for (const k of Object.keys(a)) {
    const ja = JSON.stringify(canon(a[k])), jb = JSON.stringify(canon(b[k]))
    if (ja !== jb) {
      // 하위 키까지
      const av = a[k], bv = b[k]
      if (av && bv && typeof av === 'object' && typeof bv === 'object' && !Array.isArray(av)) {
        const sub = new Set([...Object.keys(av as object), ...Object.keys(bv as object)])
        const subs: string[] = []
        for (const s of sub) {
          const x = JSON.stringify(canon((av as Record<string, unknown>)[s])), y = JSON.stringify(canon((bv as Record<string, unknown>)[s]))
          if (x !== y) subs.push(`${s}: ${x?.slice(0, 120)} -> ${y?.slice(0, 120)}`)
        }
        out.push(`${k}{${subs.slice(0, 6).join(' | ')}}`)
      } else out.push(`${k}: ${ja?.slice(0, 200)} -> ${jb?.slice(0, 200)}`)
    }
  }
  return out
}

}

function roundTrip(g: GameState): string[] {
  const raw = serialize(g)
  const back = deserialize(raw, CONTENT)
  syncHome(g) // deserialize가 지도 전역을 바꾼다 (기존 L-2)
  if (!back) return ['deserialize null']
  const d = diffKeys(summarize(g) as never, summarize(back) as never)
  // 두 번째 왕복: 불러온 것을 다시 저장해도 같아야 한다 (안정성)
  const back2 = deserialize(serialize(back), CONTENT)
  syncHome(g)
  const d2 = back2 ? diffKeys(summarize(back) as never, summarize(back2) as never).map((x) => `2nd:${x}`) : ['2nd deserialize null']
  return [...d, ...d2]
}

    if (r < 0.86) return act('takeChest/putPantry', () => { const it = randomInvItem(); S().putPantry(it as never); S().takePantry(it as never); S().takeChest(it as never) })
    if (r < 0.90) return act('discard', () => { const it = randomInvItem(); S().discard(it as never, 1) })
    if (r < 0.94) return act('feastShare', () => { const it = randomInvItem(); S().feastShare(it as never) })
    return act('heartsBoost(cheat-off)', () => {})
  }
  const freeAction = () => {
    if (rnd() < (start === 'social' ? 0.25 : 0.08)) return extraAction()
    const r = rnd()
    const g = S().game
    if (r < 0.22) { const p = pick(PLACE_IDS); const t = PLACES[p].tiles[0]; if (t) return act(`tap(${p})`, () => S().tap(t)) }
    if (r < 0.34) { const id = pick(NB); const t = npcTile(id); if (t) return act(`tapNpc(${id})`, () => S().tap(t)) }
    if (r < 0.38) { const t = { x: Math.round(g.player.x) + R(9) - 4, y: Math.round(g.player.y) + R(9) - 4 }; return act(`tapNear`, () => S().tap(t)) }
    if (r < 0.42) return act('walk', () => S().walk(R(3) - 1, R(3) - 1))
    if (r < 0.45) return act('interact', () => S().interact())
    if (r < 0.57) return frames(5 + R(20), pick([0.25, 1, 3]))
    if (r < 0.62) { const b = randomBook(); return act(`copyBook(${b})`, () => S().copyBook(b)) }
    if (r < 0.65) { const done = BOOKS.filter((b) => (g.progress[b]?.completed?.length ?? 0) > 0); if (done.length) { const b = pick(done); const c = pick(g.progress[b].completed); return act(`copyAgain(${b},${c})`, () => (S() as unknown as { copyAgain: (b: string, c: number) => void }).copyAgain(b, c)) } }
    if (r < 0.67) return act('openTrade', () => S().open({ kind: 'trade' }))
    if (r < 0.69) { const it = randomInvItem(); return act(`sellItem(${it})`, () => S().sellItem(it as never)) }
    if (r < 0.70) return act('startReading', () => S().startReading())
    if (r < 0.71) return act('openTravel', () => S().open({ kind: 'travel' }))
    if (r < 0.72) return act('eat', () => S().eat())
    if (r < 0.73) return act('lookAround', () => S().lookAround())
    if (r < 0.74) { const id = pick(NB); return act(`talkTo(${id})`, () => S().talkTo(id)) }
    if (r < 0.75) { const pl = pick(['tea', 'sunset', 'walk'] as const); return act(`goDate(${pl})`, () => S().goDate(pl)) }
    if (r < 0.76) return act('askBaby', () => S().askBaby())
    if (r < 0.77) return act('openLibrary', () => S().open({ kind: 'library' }))
    if (r < 0.78) { const b = pick(BOOKS); return act(`seal(${b})`, () => S().seal(b)) }
    if (r < 0.79) { const it = randomInvItem(); return act(`buyRareItem`, () => S().buyRareItem(it as never)) }
    if (r < 0.80) return act('drink', () => S().drink())
    if (r < 0.81) return act('blanket', () => S().blanket())
    if (r < 0.82) { const t = { x: Math.round(g.player.x), y: Math.round(g.player.y) }; const c = pick(['herb', 'bean'] as const); return act(`plantAt`, () => { S().plantAt(t, c); S().waterAt(t); S().harvestAt(t) }) }
    if (r < 0.83) return act('kidTime', () => S().open({ kind: 'kidTime' }))
    if (r < 0.84) return act('travelTo', () => S().travelTo(rnd() < 0.5 ? 'newland' : 'village'))
    if (r < 0.85) return act('cycleTimeSpeed', () => S().cycleTimeSpeed())
    if (r < 0.86) return act('reload', () => { const g2 = deserialize(serialize(S().game), CONTENT); if (g2) S().load(g2) })
    if (r < 0.87) return act('family', () => S().open({ kind: 'family' }))
    return frames(3, 1)
  }

  const endOfDay = () => {
    if (weddingToday(S().game) && rnd() < 0.9) {
      act('goWedding', () => {
        for (let i = 0; i < 30 && S().modal; i++) S().closeModal()
        if (S().game.map === 'newland') S().travelTo('village')
        S().tap({ x: 25, y: 21 })
        for (let i = 0; i < 1400 && S().game.romance?.stage === 'engaged'; i++) { if (S().modal) { if (S().modal!.kind === 'scene') S().nextScene(); else S().closeModal() } S().frame(1) }
      })
    }
    // 잠: 침대로 가서 창을 열거나, 바로 창을 열어 잔다
    act('goBed', () => { if (S().modal) S().closeModal(); S().open({ kind: 'review', pieceId: null }) })
    if (S().modal?.kind === 'review') act('sleep', () => { S().sleep(); stats.sleeps++ })
  }

  for (let d = 0; d < days; d++) {
    const dayAt = S().game.clock.day
    for (let k = 0; k < stepsPerDay; k++) {
      if (stopAt !== undefined && step >= stopAt) break
      const s = S()
      if (s.trip) { act('finishTripBoard', () => S().finishTripBoard(rnd() < 0.5 ? [] : [{ kind: 'coins', n: 3 } as never])); continue }
      if (s.copyResult?.length) { act('dismissCopyResult', () => S().dismissCopyResult()); continue }
      if (s.piecePopups?.length) { act('dismissPiecePopup', () => S().dismissPiecePopup()); continue }
      if (s.decorating) { act('stopDecorate', () => S().stopDecorate()); continue }
      if (s.modal) modalAction(); else freeAction()
      // 창이 오래 열린 채 고정되는지
      const mk = S().modal ? JSON.stringify(S().modal).slice(0, 200) : ''
      if (mk && mk === modalKey) { modalSince++ } else { modalKey = mk; modalSince = 0 }
      if (modalSince === 40) {
        // 닫기를 여러 번 눌러 본다
        for (let i = 0; i < 5; i++) act('closeModal(stuck?)', () => { S().closeModal(); S().frame(0.1) })
        if (S().modal && JSON.stringify(S().modal).slice(0, 200) === modalKey) report('stuckModal', `${S().modal!.kind}: ${modalKey}`)
      }
      if (S().modal) stats.modalsSeen.set(S().modal!.kind, (stats.modalsSeen.get(S().modal!.kind) ?? 0) + 1)
      { const gg = S().game; if (gg.romance?.stage) reached.add('romance:' + gg.romance.stage); if (gg.child) reached.add('child'); if ((gg.siblings ?? []).length) reached.add('siblings'); if (gg.homeLevel) reached.add('home' + gg.homeLevel); if (gg.companion) reached.add('pet'); if (gg.flags.newlandGift) reached.add('newland'); if ((gg.clubs ?? []).length) reached.add('club') }
      if (step % 25 === 0) {
        const g = S().game
        for (const x of checkInvariants(g, prevDay)) report('invariant', x)
        prevDay = Math.max(prevDay, g.clock.day)
        if (step % 100 === 0) {
          stats.roundTrips++
          try { for (const x of roundTrip(g)) report('roundtrip', x) } catch (e) { report('roundtripException', String((e as Error).stack).slice(0, 400)) }
        }
      }
      vi.runOnlyPendingTimers()
    }
    if (stopAt !== undefined && step >= stopAt) break
    // 하루 끝: 늦게까지 시간을 흘리고 잔다
    if (rnd() < 0.5) frames(10, 30)
    endOfDay()
    vi.runOnlyPendingTimers()
    const g = S().game
    for (const x of checkInvariants(g, prevDay)) report('invariant', x)
    if (g.clock.day === dayAt) {
      // 잠이 날짜를 넘기지 못했다
      report('noDayAdvance', `day ${dayAt} modal=${S().modal?.kind} trip=${!!S().trip}`)
      // 강제로 진행: 창 모두 닫기 반복 후 다시
      for (let i = 0; i < 10 && S().modal; i++) act('closeModal(force)', () => S().closeModal())
      endOfDay()
    }
    prevDay = Math.max(prevDay, S().game.clock.day)
    stats.roundTrips++
    try { for (const x of roundTrip(S().game)) report('roundtrip', x) } catch (e) { report('roundtripException', String((e as Error).stack).slice(0, 400)) }
  }
  Math.random = origRandom
  const g = S().game
  return {
    seed, start, steps: step, day: g.clock.day, ...stats,
    modalsSeen: Object.fromEntries(stats.modalsSeen), actionsSeen: Object.fromEntries(stats.actionsSeen),
    undefinedScenes: [...undefinedScenes], reached: [...reached], whyCord: Object.fromEntries(whyCord),
    end: { coins: g.coins, books: BOOKS.filter((b) => (g.progress[b]?.completed?.length ?? 0) > 0).length, shelved: Object.keys(g.shelved).length, collected: g.collected.length, child: !!g.child, romance: g.romance?.stage ?? null, homeLevel: g.homeLevel, persons: Object.keys((g.gen as unknown as { persons?: object })?.persons ?? {}).length },
    fullLog,
  }
}

it('r4sim fuzz', async () => {
  vi.useFakeTimers({ toFake: ['setTimeout', 'clearTimeout', 'setInterval', 'clearInterval'] })
  const seeds = Number(process.env.SEEDS ?? 8)
  const days = Number(process.env.DAYS ?? 300)
  const spd = Number(process.env.STEPS_PER_DAY ?? 60)
  const startMode = process.env.START ?? 'mix'
  const seedBase = Number(process.env.SEED_BASE ?? 1)
  const replay = process.env.REPLAY // "seed:step:start"
  const tag = process.env.TAG ?? 'run'
  const vf = `${OUT}/${tag}-violations.jsonl`
  writeFileSync(vf, '')
  const results: unknown[] = []
  const t0 = Date.now()
  if (replay) {
    const [s, st, sm] = replay.split(':')
    const r = await runSeed(Number(s), days, (sm as 'new' | 'late' | 'social') ?? 'new', spd, (v) => appendFileSync(vf, JSON.stringify(v) + '\n'), Number(st))
    writeFileSync(`${OUT}/${tag}-replay-log.txt`, r.fullLog.join('\n'))
    results.push({ ...r, fullLog: undefined })
  } else {
    for (let i = 0; i < seeds; i++) {
      const seed = seedBase + i
      const start = startMode === 'mix' ? (['new', 'late', 'social'] as const)[i % 3] : (startMode as 'new' | 'late' | 'social')
      const ts = Date.now()
      const r = await runSeed(seed, days, start, spd, (v) => appendFileSync(vf, JSON.stringify(v) + '\n'))
      results.push({ ...r, fullLog: undefined, ms: Date.now() - ts })
      writeFileSync(`${OUT}/${tag}-results.json`, JSON.stringify(results, null, 1))
    }
  }
  writeFileSync(`${OUT}/${tag}-results.json`, JSON.stringify({ totalMs: Date.now() - t0, results }, null, 1))
  vi.useRealTimers()
  expect(true).toBe(true)
})
