beforeEach(() => setPeopleData(PEOPLE))

type Facing = 'up' | 'down' | 'left' | 'right'
function furnish(s: GameState, pieces: [ItemId, number, number, Facing?][]): GameState {
  syncHome(s)
  let room: Furniture[] = [...s.room]
  for (const [item, x, y, facing] of pieces) {
    const f = placement(room, item, { x, y }, facing)
    if (!f) throw new Error(`cannot place ${item} ${x},${y}`)
    room = [...room, f]
    syncHome({ ...s, room })
  }
  return { ...s, room }
}

describe('liveSpaces memo vs walls', () => {
  it('벽을 세워 의자 곁 서는 칸을 모두 막아도 memo가 옛 결과를 돌려주는가', () => {
    const s0 = settle(newGame(CONTENT, { look: 'f', name: '하늘' } as never), CONTENT)
    let s: GameState = { ...s0, homeLevel: 1, coins: 500, flags: { ...s0.flags, growthLegacy: 1 }, player: { ...s0.player, ...HOME_ENTRY }, scenes: [] }
    s.progress.phm.completed = [1]
    syncHome(s)
    // 가족 쉼터: 의자 하나 (family는 앉을 칸만 있으면 된다)
    const log: string[] = []
    let chosen: GameState | null = null
    // 작업실 안 아무 바닥이나 시도
    outer: for (let y = 108; y <= 117; y++) for (let x = 6; x <= 30; x++) {
      try {
        const t = furnish(s, [['stool', x, y, 'down']])
        const f = t.room[t.room.length - 1]
        const n = setSpace(t, f, 'family')

      } catch { /* next */ }

    let g = chosen!
    const stool = g.room[g.room.length - 1]
    expect(liveSpaces(g).length).toBe(1) // memo 생성

    const around = [{ x: stool.x, y: stool.y + 1 }, { x: stool.x - 1, y: stool.y }, { x: stool.x + 1, y: stool.y }, { x: stool.x, y: stool.y - 1 }]
    for (const t of around) {
      const why = canBuildWall(g, t)
      log.push(`wall ${t.x},${t.y}: ${why ?? 'ok'}`)

    syncHome(g)
    const memo = liveSpaces(g).length
    const fresh = liveSpaces({ ...g, room: [...g.room] }).length
    log.push(`after walls: memo ${memo}, fresh ${fresh}, spaceReady ${spaceReady(g.room, g.spaces![0])}`)
