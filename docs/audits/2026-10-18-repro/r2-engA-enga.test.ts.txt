function furnish(s: GameState, pieces: [ItemId, number, number, any?][]): GameState {
  syncHome(s)
  let room: Furniture[] = [...s.room]
  for (const [item, x, y, facing] of pieces) { const f = placement(room, item, { x, y }, facing); expect(f).not.toBeNull(); room = [...room, f!]; syncHome({ ...s, room }) }
  return { ...s, room }
}
