it('duplicate tiles', () => {
  let total = 0, dup = 0; const ex: string[] = []
  for (const b of BOOKS) for (let c = 1; c < 200; c++) {
    const vs = chapterText(b, c); if (!vs.length) break
    for (const v of vs) { const t = tilesFor(v.text, 3, 1, b, c, v.verse); if (!t) continue; total++
      if (new Set(t.pieces).size !== t.pieces.length) { dup++; if (ex.length < 5) ex.push(`${b} ${c}:${v.verse} ${t.pieces.join('|')}`) } }
  }
  console.log('tiles total', total, 'dup', dup, ex)
})
